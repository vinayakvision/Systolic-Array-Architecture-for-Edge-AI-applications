    input  logic        rst,

    input  logic [511:0] weight_flat,
    output logic [63:0] Re1, Re2, Re3, Re4
);
    //logic [31:0] w [0:15];
    logic [63:0] C [0:3][0:3];
    logic [31:0] E [0:3][0:3];

    /* -------- Weight unpacking (explicit, Genus-safe) -------- */
    /*assign w[0]  = weight_flat[ 31:  0];
    assign w[1]  = weight_flat[ 63: 32];
    assign w[2]  = weight_flat[ 95: 64];
    assign w[3]  = weight_flat[127: 96];

    assign w[4]  = weight_flat[159:128];
    assign w[5]  = weight_flat[191:160];

    assign w[7]  = weight_flat[255:224];

    assign w[11] = weight_flat[383:352];

    assign w[12] = weight_flat[415:384];
    assign w[13] = weight_flat[447:416];
    assign w[14] = weight_flat[479:448];
    assign w[15] = weight_flat[511:480];*/

    /* -------- Row 0 -------- */
    block P1  (64'd0,   weight_flat[ 31:  0],  inP1,    clk, rst, E[0][0], C[0][0]);
    block P2  (64'd0,   weight_flat[ 63: 32],  E[0][0], clk, rst, E[0][1], C[0][1]);
    block P3  (64'd0,   weight_flat[ 95: 64],  E[0][1], clk, rst, E[0][2], C[0][2]);
    block P4  (64'd0,   weight_flat[127: 96],  E[0][2], clk, rst, E[0][3], C[0][3]);

    /* -------- Row 1 -------- */
    block P5  (C[0][0], weight_flat[159:128],  inP5,    clk, rst, E[1][0], C[1][0]);
    block P6  (C[0][1], weight_flat[191:160],  E[1][0], clk, rst, E[1][1], C[1][1]);
    block P7  (C[0][2], weight_flat[223:192],  E[1][1], clk, rst, E[1][2], C[1][2]);
    block P8  (C[0][3], weight_flat[255:224],  E[1][2], clk, rst, E[1][3], C[1][3]);

    /* -------- Row 2 -------- */
    block P9  (C[1][0], weight_flat[287:256],  inP9,    clk, rst, E[2][0], C[2][0]);
    block P10 (C[1][1], weight_flat[319:288],  E[2][0], clk, rst, E[2][1], C[2][1]);
    block P11 (C[1][2], weight_flat[351:320], E[2][1], clk, rst, E[2][2], C[2][2]);
    block P12 (C[1][3], weight_flat[383:352], E[2][2], clk, rst, E[2][3], C[2][3]);

    /* -------- Row 3 -------- */
    block P13 (C[2][0], weight_flat[415:384], inP13,   clk, rst, E[3][0], Re1);
    block P14 (C[2][1], weight_flat[447:416], E[3][0], clk, rst, E[3][1], Re2);
    block P15 (C[2][2], weight_flat[479:448], E[3][1], clk, rst, E[3][2], Re3);
    block P16 (C[2][3], weight_flat[511:480], E[3][2], clk, rst, E[3][3], Re4);

endmodule
